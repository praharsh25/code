import pandas as pd


df = pd.read_csv(r"C:\Users\YourName\Downloads\Disease_symptom_and_patient_profile_dataset.csv")

print("Dataset:")
print(df.head())

features = ["Fever", "Cough", "Fatigue", "Difficulty Breathing",
            "Age", "Gender", "Blood Pressure", "Cholesterol Level"]

rules = []

for _, row in df.iterrows():
    conditions = set(str(row[col]).lower() for col in features)
    disease = str(row["Disease"])
    rules.append((conditions, disease))

patient = df.iloc[0]

facts = set(str(patient[col]).lower() for col in features)

print("\nPatient Facts:")
print(facts)

def forward_chaining(facts):
    for conditions, disease in rules:
        if conditions.issubset(facts):
            return disease
    return None

result = forward_chaining(facts)

print("\nForward Chaining Result:", result)

def backward_chaining(goal, facts):
    for conditions, disease in rules:
        if disease == goal and conditions.issubset(facts):
            return True
    return False

if result:
    print("Backward Chaining Result:",
          backward_chaining(result, facts))
else:
    print("No matching disease found")
