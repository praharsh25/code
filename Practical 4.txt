
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

sem_t fork[5], room;

void *philosopher(void *arg)
{
    int id = *(int *)arg;
    int left = id;
    int right = (id + 1) % 5;

    for(int i = 0; i < 2; i++)
    {
        printf("Philosopher %d is thinking\n", id);

        sem_wait(&room);

        sem_wait(&fork[left]);
        sem_wait(&fork[right]);

        printf("Philosopher %d is eating\n", id);
        sleep(1);

        sem_post(&fork[right]);
        sem_post(&fork[left]);

        sem_post(&room);
    }

    return NULL;
}

int main()
{
    pthread_t t[5];
    int id[5];

    sem_init(&room, 0, 4);

    for(int i = 0; i < 5; i++)
        sem_init(&fork[i], 0, 1);

    for(int i = 0; i < 5; i++)
    {
        id[i] = i;
        pthread_create(&t[i], NULL, philosopher, &id[i]);
    }

    for(int i = 0; i < 5; i++)
        pthread_join(t[i], NULL);

    return 0;
}



Run command:
nano dining.c
gcc dining.c -o dining -pthread
./dining
