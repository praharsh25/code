
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#include <ctype.h>

int main(int argc, char *argv[])
{
    char s[1000], result[200];
    int fd, ch, words = 0, lines = 0, inword = 0;
    int chars = 0;

    mkfifo("fifo1", 0666);
    mkfifo("fifo2", 0666);

    if(argc < 2)
    {
        printf("Use: ./fifo sender OR ./fifo receiver\n");
        return 1;
    }

    if(strcmp(argv[1], "sender") == 0)
    {
        fd = open("fifo1", O_WRONLY);

        printf("Enter sentences (END to stop):\n");

        while(fgets(s, sizeof(s), stdin))
        {
            if(strcmp(s, "END\n") == 0)
                break;

            write(fd, s, strlen(s));
        }

        close(fd);

        fd = open("fifo2", O_RDONLY);
        read(fd, result, sizeof(result)-1);
        result[sizeof(result)-1] = '\0';

        printf("\nResult from second process:\n%s", result);
        close(fd);
    }
    else if(strcmp(argv[1], "receiver") == 0)
    {
        fd = open("fifo1", O_RDONLY);

        FILE *fp = fopen("output.txt", "w");

        while((ch = read(fd, s, sizeof(s))) > 0)
        {
            for(int i = 0; i < ch; i++)
            {
                chars++;

                if(s[i] == '\n')
                    lines++;

                if(isspace((unsigned char)s[i]))
                    inword = 0;
                else if(!inword)
                {
                    words++;
                    inword = 1;
                }
            }
        }

        close(fd);

        fprintf(fp, "Characters: %d\nWords: %d\nLines: %d\n",
                chars, words, lines);
        fclose(fp);

        fp = fopen("output.txt", "r");
        fd = open("fifo2", O_WRONLY);

        while(fgets(s, sizeof(s), fp))
            write(fd, s, strlen(s));

        fclose(fp);
        close(fd);
    }

    return 0;
}




Run commands:
Step 1: Create file
nano fifo.c

Step 2: Compile
gcc fifo.c -o fifo


Step 3: Open Terminal 1 and run the receiver
./fifo receiver


Step 4: Open Terminal 2 and run the sender
./fifo sender


Step 5: Enter sentences in Terminal 2
Enter sentences (END to stop):

Operating System
Linux Programming
Interprocess Communication
END


Step 6: Output in Terminal 2
Result from second process:

Characters: 59
Words: 7
Lines: 3

Step 7: Check the output file
