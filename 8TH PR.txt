
1)	Create Procedure For Incrementing Salary 
Create Table-
CREATE TABLE EMPLOYEE (
EMP_ID NUMBER PRIMARY KEY,
EMP_NAME VARCHAR2(50),
SALARY NUMBER
);

Insert value-
INSERT INTO EMPLOYEE VALUES (1, 'Amit', 30000);
INSERT INTO EMPLOYEE VALUES (2, 'Neha', 40000);
COMMIT;

Create Procedure-
CREATE OR REPLACE PROCEDURE INCREMENT_SALARY(
E_ID IN NUMBER,
INC_AMT IN NUMBER
)
IS
BEGIN
UPDATE EMPLOYEE
SET SALARY = SALARY + INC_AMT
WHERE EMP_ID = E_ID;
DBMS_OUTPUT.PUT_LINE('Salary Updated Successfully');
END;
/

Run Procedure-
SET SERVEROUTPUT ON;

BEGIN
INCREMENT_SALARY(1, 5000);
END;
/

2)	Calculate annual salary(Function)
CREATE OR REPLACE FUNCTION ANNUAL_SAL(
    E_ID IN NUMBER
)
RETURN NUMBER
IS
    AN_SAL NUMBER;
BEGIN
    SELECT SALARY * 12 INTO AN_SAL
    FROM EMPLOYEE
    WHERE EMP_ID = E_ID;

    RETURN AN_SAL;
END;
/

Call Function :
SET SERVEROUTPUT ON;

DECLARE
    RESULT NUMBER;
BEGIN
    RESULT := ANNUAL_SAL(1);
    DBMS_OUTPUT.PUT_LINE('Annual Salary = ' || RESULT);
END;
/
